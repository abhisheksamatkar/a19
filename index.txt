//////////////////////////////4.b) Design an AJAX-based webpage for an online feedback system that submits and
displays user feedback dynamically without reloading the page.


Feedback.html


<!DOCTYPE html>
<html>
<head>
    <title>Feedback System</title>
    <style>
        body { font-family: Arial; max-width: 500px; margin: 50px auto; padding: 20px; }
        .feedback { background: #f9f9f9; margin: 10px 0; padding: 10px; border-left: 4px solid blue; }
        input, textarea, button { width: 100%; margin: 5px 0; padding: 8px; }
        button { background: blue; color: white; border: none; cursor: pointer; }
        .star { color: gold; }
    </style>
</head>
<body>
    <h2>📝 Student Feedback</h2>
    
    <!-- Form -->
    <input type="text" id="name" placeholder="Your Name">
    <select id="rating">
        <option value="5">★★★★★ Excellent</option>
        <option value="4">★★★★☆ Good</option>
        <option value="3">★★★☆☆ Average</option>
    </select>
    <textarea id="msg" rows="3" placeholder="Your feedback..."></textarea>
    <button onclick="addFeedback()">Submit Feedback</button>
    
    <hr>
    <h3>All Feedbacks <span id="count"></span></h3>
    <div id="list"></div>

    <script>
        let feedbacks = [];
        
        // Show all feedbacks
        function show() {
            let html = '';
            for(let i = 0; i < feedbacks.length; i++) {
                let stars = '★'.repeat(feedbacks[i].rating) + '☆'.repeat(5-feedbacks[i].rating);
                html += '<div class="feedback">' +
                        '<strong>' + feedbacks[i].name + '</strong> ' +
                        '<span class="star">' + stars + '</span><br>' +
                        feedbacks[i].msg + '<br>' +
                        '<small>' + feedbacks[i].date + '</small>' +
                        ' <button onclick="del(' + i + ')">Delete</button>' +
                        '</div>';
            }
            document.getElementById('list').innerHTML = html || '<p>No feedback yet</p>';
            document.getElementById('count').innerHTML = '(' + feedbacks.length + ')';
        }
        
        // Add feedback (AJAX style - no page reload)
        function addFeedback() {
            let name = document.getElementById('name').value;
            let rating = document.getElementById('rating').value;
            let msg = document.getElementById('msg').value;
            
            if(name === '' || msg === '') {
                alert('Please fill all fields');
                return;
            }
            
            // Add to array
            feedbacks.unshift({
                name: name,
                rating: parseInt(rating),
                msg: msg,
                date: new Date().toLocaleString()
            });
            
            // Save to storage
            localStorage.setItem('feedbacks', JSON.stringify(feedbacks));
            
            // Clear form
            document.getElementById('name').value = '';
            document.getElementById('msg').value = '';
            document.getElementById('rating').value = '5';
            
            // Update display (NO PAGE RELOAD!)
            show();
            alert('Feedback submitted!');
        }
        
        // Delete feedback
        function del(index) {
            feedbacks.splice(index, 1);
            localStorage.setItem('feedbacks', JSON.stringify(feedbacks));
            show();
        }
        
        // Load saved feedbacks when page opens
        let saved = localStorage.getItem('feedbacks');
        if(saved) {
            feedbacks = JSON.parse(saved);
        } else {
            // Sample data
            feedbacks = [
                {name: "Rahul", rating: 5, msg: "Great college!", date: "Today"},
                {name: "Priya", rating: 4, msg: "Good teachers", date: "Yesterday"}
            ];
        }
        show();
    </script>
</body>
</html>

import React, { useState } from 'react';

function App() {
  // Student tuple data [roll, name, marks]
  const students = [
    ["101", "Rahul", 85],
    ["102", "Priya", 92],
    ["103", "Amit", 67],
    ["104", "Neha", 78],
    ["105", "Raj", 55]
  ];

  const [roll, setRoll] = useState("");
  const [result, setResult] = useState(null);

  const search = () => {
    const found = students.find(s => s[0] === roll);
    found ? setResult(found) : alert("Not Found");
  };

  return (
    <div style={{textAlign: "center", marginTop: 30}}>
      <h2>Student Result</h2>
      <input placeholder="Enter Roll No" onChange={e => setRoll(e.target.value)} />
      <button onClick={search}>Search</button>

      {result && (
        <div style={{border: "2px solid green", padding: 20, margin: 20}}>
          <p><b>Name:</b> {result[1]}</p>
          <p><b>Marks:</b> {result[2]}</p>
          <p><b>Status:</b> {result[2] >= 50 ? "Pass" : "Fail"}</p>
        </div>
      )}

      <h3>All Students</h3>
      <table border="1" style={{margin: "auto"}}>
        <thead><tr><th>Roll</th><th>Name</th><th>Marks</th></tr></thead>
        <tbody>
          {students.map(s => <tr><td>{s[0]}</td><td>{s[1]}</td><td>{s[2]}</td></tr>)}
        </tbody>
      </table>
    </div>
  );
}

export default App;

/////////////////////////////////////////////10.a) Design a webpage in React for managing student course registrations with features
to add, view, and update course details dynamically.

import React, { useState } from 'react';

function App() {
  // Course list
  const [courses, setCourses] = useState([
    { id: 1, name: "JavaScript", fee: 500 },
    { id: 2, name: "React JS", fee: 700 },
    { id: 3, name: "Python", fee: 450 }
  ]);

  const [name, setName] = useState('');
  const [fee, setFee] = useState('');
  const [editId, setEditId] = useState(null);

  // Add course
  const addCourse = () => {
    if(name && fee) {
      const newCourse = { id: Date.now(), name: name, fee: parseInt(fee) };
      setCourses([...courses, newCourse]);
      setName('');
      setFee('');
    }
  };

  // Delete course
  const deleteCourse = (id) => {
    setCourses(courses.filter(c => c.id !== id));
  };

  // Edit course
  const editCourse = (course) => {
    setName(course.name);
    setFee(course.fee);
    setEditId(course.id);
  };

  // Update course
  const updateCourse = () => {
    const updated = courses.map(c => 
      c.id === editId ? { ...c, name: name, fee: parseInt(fee) } : c
    );
    setCourses(updated);
    setName('');
    setFee('');
    setEditId(null);
  };

  return (
    <div style={{ fontFamily: 'Arial', maxWidth: '600px', margin: '50px auto', padding: '20px' }}>
      <h1 style={{ textAlign: 'center' }}>📚 Course Registration</h1>
      
      {/* Form */}
      <div style={{ background: '#f0f0f0', padding: '20px', borderRadius: '10px' }}>
        <input 
          type="text" 
          placeholder="Course Name" 
          value={name} 
          onChange={(e) => setName(e.target.value)}
          style={{ padding: '10px', margin: '5px', width: '200px' }}
        />
        <input 
          type="number" 
          placeholder="Fee ($)" 
          value={fee} 
          onChange={(e) => setFee(e.target.value)}
          style={{ padding: '10px', margin: '5px', width: '100px' }}
        />
        
        {editId ? (
          <button onClick={updateCourse} style={{ padding: '10px', margin: '5px', background: 'orange', color: 'white', border: 'none', cursor: 'pointer' }}>Update</button>
        ) : (
          <button onClick={addCourse} style={{ padding: '10px', margin: '5px', background: 'green', color: 'white', border: 'none', cursor: 'pointer' }}>Add</button>
        )}
      </div>
      
      {/* Course List */}
      <h3>Course List ({courses.length})</h3>
      <table style={{ width: '100%', borderCollapse: 'collapse' }}>
        <thead>
          <tr style={{ background: '#333', color: 'white' }}>
            <th style={{ padding: '10px' }}>Name</th>
            <th style={{ padding: '10px' }}>Fee</th>
            <th style={{ padding: '10px' }}>Action</th>
           </tr>
        </thead>
        <tbody>
          {courses.map(c => (
            <tr key={c.id} style={{ textAlign: 'center' }}>
              <td style={{ padding: '8px' }}>{c.name}</td>
              <td>${c.fee}</td>
              <td>
                <button onClick={() => editCourse(c)} style={{ background: 'orange', color: 'white', border: 'none', padding: '5px 10px', margin: '2px', cursor: 'pointer' }}>Edit</button>
                <button onClick={() => deleteCourse(c.id)} style={{ background: 'red', color: 'white', border: 'none', padding: '5px 10px', margin: '2px', cursor: 'pointer' }}>Del</button>
              </td>
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  );
}

export default App;

npm start

app.use(express.static('../frontend'));

// Read/Write JSON file
const readData = () => JSON.parse(fs.readFileSync('./data.json', 'utf8'));
const saveData = (data) => fs.writeFileSync('./data.json', JSON.stringify(data, null, 2));

// Register
app.post('/api/register', (req, res) => {
    const { name, email, password } = req.body;
    const data = readData();
    
    if (data.users.find(u => u.email === email)) {
        return res.json({ ok: false, msg: "Email exists!" });
    }
    
    data.users.push({ id: data.users.length + 1, name, email, password });
    saveData(data);
    res.json({ ok: true, msg: "Registered!" });
});

// Login
app.post('/api/login', (req, res) => {
    const { email, password } = req.body;
    const data = readData();
    const user = data.users.find(u => u.email === email && u.password === password);
    
    if (user) {
        res.json({ ok: true, user: { id: user.id, name: user.name, email: user.email } });
    } else {
        res.json({ ok: false, msg: "Invalid credentials!" });
    }
});

// Get products
app.get('/api/products', (req, res) => {
    const data = readData();
    const { search, category } = req.query;
    let products = data.products;
    
    if (search) {
        products = products.filter(p => p.name.toLowerCase().includes(search.toLowerCase()));
    }
    if (category && category !== 'All') {
        products = products.filter(p => p.category === category);
    }
    
    res.json({ ok: true, products });
});

// Get single product
app.get('/api/products/:id', (req, res) => {
    const data = readData();
    const product = data.products.find(p => p.id == req.params.id);
    res.json({ ok: true, product });
});

// Get categories
app.get('/api/categories', (req, res) => {
    const data = readData();
    const categories = [...new Set(data.products.map(p => p.category))];
    res.json(categories);
});

// Get cart
app.get('/api/cart', (req, res) => {
    const data = readData();
    const userCart = data.cart.filter(c => c.userId == req.query.userId);
    const cartWithProducts = userCart.map(c => ({
        ...c,
        product: data.products.find(p => p.id === c.productId)
    }));
    res.json(cartWithProducts);
});

// Add to cart
app.post('/api/cart', (req, res) => {
    const { userId, productId } = req.body;
    const data = readData();
    
    const existing = data.cart.find(c => c.userId == userId && c.productId == productId);
    if (existing) {
        existing.quantity += 1;
    } else {
        data.cart.push({ id: Date.now(), userId, productId, quantity: 1 });
    }
    
    saveData(data);
    res.json({ ok: true, msg: "Added to cart!" });
});

// Update cart quantity
app.put('/api/cart/:id', (req, res) => {
    const data = readData();
    const item = data.cart.find(c => c.id == req.params.id);
    if (item) {
        item.quantity = req.body.quantity;
        saveData(data);
    }
    res.json({ ok: true });
});

// Remove from cart
app.delete('/api/cart/:id', (req, res) => {
    const data = readData();
    data.cart = data.cart.filter(c => c.id != req.params.id);
    saveData(data);
    res.json({ ok: true });
});

// Place order
app.post('/api/orders', (req, res) => {
    const { userId, items, total } = req.body;
    const data = readData();
    
    data.orders.push({
        id: data.orders.length + 1,
        userId,
        items,
        total,
        status: "Processing",
        date: new Date().toLocaleString()
    });
    
    data.cart = data.cart.filter(c => c.userId != userId);
    saveData(data);
    res.json({ ok: true, msg: "Order placed!" });
});

// Get orders
app.get('/api/orders', (req, res) => {
    const data = readData();
    const userOrders = data.orders.filter(o => o.userId == req.query.userId).reverse();
    res.json(userOrders);
});

// Home page
app.get('/', (req, res) => {
    res.sendFile(path.join(__dirname, '../frontend/index.html'));
});

app.listen(5000, () => {
    console.log("Server running on http://localhost:5000");

async function loadCart() {
    const res = await fetch(`/api/cart?userId=${user.id}`);
    const cart = await res.json();
    
    if(cart.length === 0) {
        document.getElementById('cartItems').innerHTML = '<p>Cart is empty!</p>';
        document.getElementById('cartTotal').innerText = '0';
        return;
    }
    
    let total = 0;
    document.getElementById('cartItems').innerHTML = cart.map(c => {
        total += c.product.price * c.quantity;
        return `
            <div class="cart-item">
                <div style="font-size:40px;">${c.product.image}</div>
                <div class="info">
                    <strong>${c.product.name}</strong>
                    <p>₹${c.product.price.toLocaleString()} x ${c.quantity} = ₹${(c.product.price * c.quantity).toLocaleString()}</p>
                </div>
                <div>
                    <button class="qty-btn" onclick="updateCart(${c.id}, ${c.quantity - 1})">-</button>
                    ${c.quantity}
                    <button class="qty-btn" onclick="updateCart(${c.id}, ${c.quantity + 1})">+</button>
                    <button onclick="removeCart(${c.id})" class="danger" style="width:auto;">X</button>
                </div>
            </div>
        `;
    }).join('');
    
    document.getElementById('cartTotal').innerText = total.toLocaleString();
}

async function updateCart(cartId, qty) {
    if(qty < 1) return removeCart(cartId);
    await fetch(`/api/cart/${cartId}`, {
        method: 'PUT',
        headers: {'Content-Type': 'application/json'},
        body: JSON.stringify({ quantity: qty })
    });
    loadCart();
    updateCartCount();
}

async function removeCart(cartId) {
    await fetch(`/api/cart/${cartId}`, { method: 'DELETE' });
    loadCart();
    updateCartCount();
}

async function updateCartCount() {
    const res = await fetch(`/api/cart?userId=${user.id}`);
    const cart = await res.json();
    const count = cart.reduce((s, c) => s + c.quantity, 0);
    document.getElementById('cartCount').innerText = count;
}

// ========== ORDERS ==========
async function placeOrder() {
    const res = await fetch(`/api/cart?userId=${user.id}`);
    const cart = await res.json();
    
    if(cart.length === 0) return alert('Cart is empty!');
    if(!confirm('Confirm order?')) return;
    
    const items = cart.map(c => ({ productId: c.productId, name: c.product.name, quantity: c.quantity, price: c.product.price }));
    const total = items.reduce((s, i) => s + (i.price * i.quantity), 0);
    
    await fetch('/api/orders', {
        method: 'POST',
        headers: {'Content-Type': 'application/json'},
        body: JSON.stringify({ userId: user.id, items, total })
    });
    
    updateCartCount();
    alert('Order placed successfully!');
    showPage('orders');
}

                    <p>${o.items.map(i => `${i.name} x${i.quantity}`).join(', ')}</p>

                    <p>Status: <span style="color:orange;">${o.status}</span> | ${o.date}</p>
                </div>
            </div>
        `).join('');
}

    const data = readData();

    const cartWithItems = userCart.map(c => ({
        ...c,
        item: data.menu.find(m => m.id === c.menuId)
    }));
    res.json(cartWithItems);
});

app.post('/api/cart', (req, res) => {
    const { userId, menuId, restaurantId } = req.body;
    const data = readData();
    
    const existing = data.cart.find(c => c.userId == userId && c.menuId == menuId);
    if (existing) {
        existing.quantity += 1;
    } else {
        data.cart.push({ id: Date.now(), userId, menuId, restaurantId, quantity: 1 });
    }
    
    saveData(data);
    res.json({ ok: true, msg: "Added to cart!" });
});

app.put('/api/cart/:id', (req, res) => {
    const data = readData();
    const item = data.cart.find(c => c.id == req.params.id);
    if (item) {
        item.quantity = req.body.quantity;
        saveData(data);
    }
    res.json({ ok: true });
});

app.delete('/api/cart/:id', (req, res) => {
    const data = readData();
    data.cart = data.cart.filter(c => c.id != req.params.id);
    saveData(data);
    res.json({ ok: true });
});

    const data = readData();

        id: data.orders.length + 1,
        userId,
        items,
        total,
        restaurantName,
        address,
        paymentMethod: paymentMethod || "Cash on Delivery",
        status: "Confirmed",
        date: new Date().toLocaleString(),
        orderDate: new Date().toISOString()
    };
    
    data.orders.push(order);
    data.cart = data.cart.filter(c => c.userId != userId);
    saveData(data);
    
    res.json({ ok: true, order, msg: "Order placed successfully!" });
});

app.get('/api/orders', (req, res) => {
    const data = readData();
    const userOrders = data.orders.filter(o => o.userId == req.query.userId).reverse();
    res.json(userOrders);
});

async function addToCart(menuId, restaurantId) {

        body: JSON.stringify({ userId: user.id, menuId, restaurantId })

async function loadCart() {
    const res = await fetch(`/api/cart?userId=${user.id}`);
    const cart = await res.json();
    
    if(cart.length === 0) {
        document.getElementById('cartItems').innerHTML = '<p style="text-align:center;padding:30px;">🛒 Cart is empty!</p>';
        document.getElementById('cartTotal').innerText = '0';
        return;
    }
    
    let total = 0;
    document.getElementById('cartItems').innerHTML = cart.map(c => {
        total += c.item.price * c.quantity;
        return `
            <div class="cart-item">
                <div class="cart-item-img">${c.item.image}</div>
                <div class="cart-item-info">
                    <strong>${c.item.name}</strong>
                    <p>₹${c.item.price} x ${c.quantity} = ₹${c.item.price * c.quantity}</p>
                </div>
                <div class="qty-control">
                    <button onclick="updateCart(${c.id}, ${c.quantity - 1})">-</button>
                    <span>${c.quantity}</span>
                    <button onclick="updateCart(${c.id}, ${c.quantity + 1})">+</button>
                    <button onclick="removeCart(${c.id})" class="secondary" style="width:auto;margin-left:10px;">🗑️</button>
                </div>
            </div>
        `;
    }).join('');
    
    document.getElementById('cartTotal').innerText = total.toLocaleString();
}

async function updateCart(cartId, qty) {
    if(qty < 1) return removeCart(cartId);
    await fetch(`/api/cart/${cartId}`, {
        method: 'PUT',
        headers: {'Content-Type': 'application/json'},
        body: JSON.stringify({ quantity: qty })
    });
    loadCart();
    updateCartCount();
}

async function removeCart(cartId) {
    await fetch(`/api/cart/${cartId}`, { method: 'DELETE' });
    loadCart();
    updateCartCount();
}

async function updateCartCount() {
    const res = await fetch(`/api/cart?userId=${user.id}`);
    const cart = await res.json();
    const count = cart.reduce((s, c) => s + c.quantity, 0);
    document.getElementById('cartCount').innerText = count;
}

async function loadCheckout() {
    const res = await fetch(`/api/cart?userId=${user.id}`);
    const cart = await res.json();
    
    if(cart.length === 0) {
        alert('Cart is empty!');
        return goHome();
    }
    
    document.getElementById('deliveryAddress').value = user.address || '';
    
    let total = 0;
    document.getElementById('checkoutItems').innerHTML = cart.map(c => {
        total += c.item.price * c.quantity;
        return `<p>${c.item.image} ${c.item.name} x ${c.quantity} = ₹${c.item.price * c.quantity}</p>`;
    }).join('');
    
    document.getElementById('checkoutTotal').innerText = total.toLocaleString();
}

async function placeOrder() {
    const res = await fetch(`/api/cart?userId=${user.id}`);
    const cart = await res.json();
    
    if(cart.length === 0) return alert('Cart is empty!');
    
    const items = cart.map(c => ({ 
        menuId: c.menuId, 
        name: c.item.name, 
        quantity: c.quantity, 
        price: c.item.price 
    }));
    const total = items.reduce((s, i) => s + (i.price * i.quantity), 0);
    
    await fetch('/api/orders', {
        method: 'POST',
        headers: {'Content-Type': 'application/json'},
        body: JSON.stringify({
            userId: user.id,
            items,
            total,
            restaurantName: currentRestaurant?.name || 'Restaurant',
            address: document.getElementById('deliveryAddress').value || user.address,
            paymentMethod: document.getElementById('paymentMethod').value
        })
    });
    
    updateCartCount();
    alert('🎉 Order placed successfully!');
    showPage('orders');
}

async function loadOrders() {
    const res = await fetch(`/api/orders?userId=${user.id}`);
    const orders = await res.json();
    
    document.getElementById('orderList').innerHTML = orders.length === 0 ? 
        '<p style="text-align:center;padding:30px;">No orders yet!</p>' :
        orders.map(o => `
            <div class="order-item">
                <div class="order-item-img">🛍️</div>
                <div class="order-item-info">
                    <strong>Order #${o.id} - ${o.restaurantName}</strong>
                    <p>${o.items.map(i => `${i.name} x${i.quantity}`).join(', ')}</p>
                    <p><strong>Total: ₹${o.total.toLocaleString()}</strong> | ${o.paymentMethod}</p>
                    <p>Status: <span style="color:#2ed573;">${o.status}</span> | ${o.date}</p>
                    <p>📍 ${o.address}</p>
                </div>
            </div>
        `).join('');
}
